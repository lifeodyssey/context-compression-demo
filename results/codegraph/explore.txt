34	def validate_coupon(coupon: Coupon, subtotal: float, is_member: bool, today: date) -> None:
35	    if coupon.expires and today > coupon.expires:
36	        raise CouponError(f"{coupon.code} expired on {coupon.expires}")
37	    if coupon.members_only and not is_member:
38	        raise CouponError(f"{coupon.code} is for members only")
39	    if subtotal < coupon.min_subtotal:
40	        raise CouponError(f"{coupon.code} needs a subtotal of at least {coupon.min_subtotal}")

43	def apply_coupon(total: float, coupon: Optional[Coupon]) -> float:
44	    if coupon is None:
45	        return total
46	    if coupon.kind == "percent":
47	        return total * (1 - coupon.value / 100)
48	    if coupon.kind == "fixed":
49	        return total - coupon.value
50	    if coupon.kind == "shipping":
51	        # free shipping is modelled as a flat discount
52	        return total - coupon.value
53	    raise CouponError(f"unknown coupon kind: {coupon.kind}")

60	        return f"${coupon.value:.2f} off orders over ${coupon.min_subtotal:.2f}"
